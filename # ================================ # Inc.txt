# ================================ # Incremental LSTM for Logical Anomaly Detection + EXPERIMENTS (ONLY ADDITIONS) # ================================ import os import numpy as np import pandas as pd import psutil import matplotlib.pyplot as plt # ✅ NEW (for graphs) from sklearn.preprocessing import StandardScaler from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, roc_auc_score import tensorflow as tf from tensorflow import keras from tensorflow.keras import layers # ---------------- # ✅ OUTPUT FOLDER (NEW) # ---------------- OUT_DIR = "experiment_outputs" os.makedirs(OUT_DIR, exist_ok=True) # ---------------- # Config (BASELINE) # ---------------- WINDOW = 10 STEP = 1 TARGET_COL = "target" BATCH_SIZE = 64 EPOCHS_PER_BATCH = 1 CHUNK_SIZE = 1000 THRESHOLD = 0.5 # ✅ NEW (for threshold experiments) LSTM_UNITS = 16 # ✅ NEW (units experiment) # ---------------- # Features # ---------------- FEATURES = [c for c in final_df.columns if c != TARGET_COL] # ---------------- # Sequence builder # ---------------- def build_sequences(final_df, features, target_col=TARGET_COL, window=WINDOW, step=STEP): X = final_df[features].values y = final_df[target_col].values X_seq, y_seq = [], [] for i in range(0, len(X) - window + 1, step): X_seq.append(X[i:i+window]) y_seq.append(y[i+window-1]) return np.array(X_seq), np.array(y_seq) # ---------------- # ✅ Safe ROC AUC (NEW) -> avoids crash if chunk has only one class # ---------------- def safe_auc(y_true, y_prob): try: if len(np.unique(y_true)) < 2: return np.nan return roc_auc_score(y_true, y_prob) except: return np.nan # ---------------- # Scale data # ---------------- scaler = StandardScaler().fit(final_df[FEATURES]) final_df_scaled = final_df.copy() final_df_scaled[FEATURES] = scaler.transform(final_df[FEATURES]) # ---------------- # CPU / Memory usage helper # ---------------- PROC = psutil.Process(os.getpid()) def get_usage(): mem_bytes = PROC.memory_info().rss mem_mb = round(mem_bytes / 1024 / 1024, 2) cpu = psutil.cpu_percent(interval=0.5) sys_mem = psutil.virtual_memory().percent return {"CPU%": cpu, "Proc_Memory_MB": mem_mb, "System_Mem%": sys_mem} # ========================================================= # ✅ EXPERIMENTS LIST (ONLY ADDITION) # ========================================================= EXPERIMENTS = [] # ✅ E1 Baseline EXPERIMENTS.append({ "name": "E1_Baseline", "WINDOW": 10, "CHUNK_SIZE": 1000, "BATCH_SIZE": 64, "LSTM_UNITS": 16, "THRESHOLD": 0.5 }) # ✅ E4 Window Ablation for w in [5, 10, 20]: EXPERIMENTS.append({ "name": f"E4_Window_{w}", "WINDOW": w, "CHUNK_SIZE": 1000, "BATCH_SIZE": 64, "LSTM_UNITS": 16, "THRESHOLD": 0.5 }) # ✅ E5 Batch Ablation for b in [16, 32, 64, 128]: EXPERIMENTS.append({ "name": f"E5_Batch_{b}", "WINDOW": 10, "CHUNK_SIZE": 1000, "BATCH_SIZE": b, "LSTM_UNITS": 16, "THRESHOLD": 0.5 }) # ✅ E6 Chunk Ablation (5000 memory proof) for ch in [500, 1000, 5000]: EXPERIMENTS.append({ "name": f"E6_Chunk_{ch}", "WINDOW": 10, "CHUNK_SIZE": ch, "BATCH_SIZE": 64, "LSTM_UNITS": 16, "THRESHOLD": 0.5 }) # ✅ E7 Units Ablation (TinyML justification) for u in [16, 32, 64]: EXPERIMENTS.append({ "name": f"E7_Units_{u}", "WINDOW": 10, "CHUNK_SIZE": 1000, "BATCH_SIZE": 64, "LSTM_UNITS": u, "THRESHOLD": 0.5 }) # ✅ E8 Threshold Ablation (Precision/Recall trade-off) for t in [0.3, 0.5, 0.7]: EXPERIMENTS.append({ "name": f"E8_Threshold_{t}", "WINDOW": 10, "CHUNK_SIZE": 1000, "BATCH_SIZE": 64, "LSTM_UNITS": 16, "THRESHOLD": t }) # ========================================================= # ✅ RUN EXPERIMENTS (ONLY ADDITION, same training logic) # ========================================================= all_summaries = [] for exp in EXPERIMENTS: # Apply experimental settings (ONLY CHANGE VALUES) WINDOW = exp["WINDOW"] CHUNK_SIZE = exp["CHUNK_SIZE"] BATCH_SIZE = exp["BATCH_SIZE"] LSTM_UNITS = exp["LSTM_UNITS"] THRESHOLD = exp["THRESHOLD"] exp_name = exp["name"] print("\n" + "="*70) print(f"✅ Running Experiment: {exp_name}") print(f"WINDOW={WINDOW}, CHUNK={CHUNK_SIZE}, BATCH={BATCH_SIZE}, UNITS={LSTM_UNITS}, THRESHOLD={THRESHOLD}") print("="*70) # Split into chunks (same as your code) chunks = [final_df_scaled.iloc[i:i+CHUNK_SIZE] for i in range(0, len(final_df_scaled), CHUNK_SIZE)] # Build model (same model, only units/window changes) model = keras.Sequential([ layers.Input(shape=(WINDOW, len(FEATURES))), layers.LSTM(LSTM_UNITS), layers.Dense(1, activation="sigmoid") ]) model.compile( optimizer=keras.optimizers.Adam(1e-3), loss="binary_crossentropy", metrics=[ "accuracy", tf.keras.metrics.Recall(name="recall"), tf.keras.metrics.Precision(name="precision"), tf.keras.metrics.AUC(name="auc") ] ) # ✅ chunk-wise logs for Farzana fluctuation graphs chunk_logs = [] # ---------------- # Incremental training with before/after resource usage (YOUR SAME LOOP) # ---------------- for i, chunk_df in enumerate(chunks): # ✅ avoid crash if chunk too small if len(chunk_df) < WINDOW: continue X_chunk, y_chunk = build_sequences(chunk_df, FEATURES, TARGET_COL, WINDOW, STEP) if len(X_chunk) == 0: continue print(f"\n--- Training on chunk {i+1}/{len(chunks)} | Shape: {X_chunk.shape} ---") usage_before = get_usage() print("Usage (before fit):", usage_before) t0 = tf.timestamp() model.fit(X_chunk, y_chunk, epochs=EPOCHS_PER_BATCH, batch_size=BATCH_SIZE, verbose=1) t1 = tf.timestamp() usage_after = get_usage() print("Usage (after fit):", usage_after) y_prob = model.predict(X_chunk, verbose=0).ravel() y_pred = (y_prob >= THRESHOLD).astype(int) metrics = { "accuracy": accuracy_score(y_chunk, y_pred), "precision": precision_score(y_chunk, y_pred, zero_division=0), "recall": recall_score(y_chunk, y_pred, zero_division=0), "f1": f1_score(y_chunk, y_pred, zero_division=0), "roc_auc": safe_auc(y_chunk, y_prob) } print("Chunk metrics:", metrics) # ✅ store chunk log row chunk_logs.append({ "chunk_id": i+1, "train_samples": len(X_chunk), "CPU_before": usage_before["CPU%"], "Mem_before_MB": usage_before["Proc_Memory_MB"], "CPU_after": usage_after["CPU%"], "Mem_after_MB": usage_after["Proc_Memory_MB"], "accuracy": metrics["accuracy"], "precision": metrics["precision"], "recall": metrics["recall"], "f1": metrics["f1"], "roc_auc": metrics["roc_auc"], "train_time_sec": float((t1 - t0).numpy()) }) # ✅ Save chunk logs CSV chunk_df_log = pd.DataFrame(chunk_logs) chunk_csv_path = os.path.join(OUT_DIR, f"{exp_name}_chunk_log.csv") chunk_df_log.to_csv(chunk_csv_path, index=False) print("\n📌 Saved chunk log:", chunk_csv_path) # ✅ Farzana fluctuation graphs (Chunk vs F1/AUC/Mem/CPU) if len(chunk_df_log) > 0: x = chunk_df_log["chunk_id"].values def save_plot(col, title): plt.figure() plt.plot(x, chunk_df_log[col].values, marker="o") plt.title(f"{title} — {exp_name}") plt.xlabel("Chunk ID") plt.ylabel(col) plt.grid(True) fig_path = os.path.join(OUT_DIR, f"{exp_name}_{col}.png") plt.savefig(fig_path, dpi=200, bbox_inches="tight") plt.show() print("✅ Saved:", fig_path) save_plot("f1", "Chunk vs F1") save_plot("roc_auc", "Chunk vs ROC-AUC") save_plot("Mem_after_MB", "Chunk vs RSS Memory (MB)") save_plot("CPU_after", "Chunk vs CPU (%)") save_plot("train_time_sec", "Chunk vs Train Time (sec)") # ---------------- # Final evaluation (YOUR SAME STYLE) # ---------------- test_size = len(final_df_scaled) # ✅ kept exactly like your code test_df_scaled = final_df_scaled.iloc[-test_size:] X_test, y_test = build_sequences(test_df_scaled, FEATURES, TARGET_COL, WINDOW, STEP) print("\n=== Usage before final test ===") usage_test_before = get_usage() print(usage_test_before) y_prob = model.predict(X_test, verbose=0).ravel() y_pred = (y_prob >= THRESHOLD).astype(int) print("=== Usage after final inference ===") usage_test_after = get_usage() print(usage_test_after) final_metrics = { "accuracy": accuracy_score(y_test, y_pred), "precision": precision_score(y_test, y_pred, zero_division=0), "recall": recall_score(y_test, y_pred, zero_division=0), "f1": f1_score(y_test, y_pred, zero_division=0), "roc_auc": safe_auc(y_test, y_prob) } print("\n=== Final Test Metrics ===") print(final_metrics) # ✅ Save summary row all_summaries.append({ "experiment": exp_name, "WINDOW": WINDOW, "CHUNK_SIZE": CHUNK_SIZE, "BATCH_SIZE": BATCH_SIZE, "LSTM_UNITS": LSTM_UNITS, "THRESHOLD": THRESHOLD, "final_accuracy": final_metrics["accuracy"], "final_precision": final_metrics["precision"], "final_recall": final_metrics["recall"], "final_f1": final_metrics["f1"], "final_auc": final_metrics["roc_auc"], "rss_before_test_MB": usage_test_before["Proc_Memory_MB"], "rss_after_test_MB": usage_test_after["Proc_Memory_MB"], "cpu_before_test_%": usage_test_before["CPU%"], "cpu_after_test_%": usage_test_after["CPU%"] }) # ✅ Save ALL experiment summary CSV summary_df = pd.DataFrame(all_summaries) summary_path = os.path.join(OUT_DIR, "ALL_EXPERIMENTS_SUMMARY.csv") summary_df.to_csv(summary_path, index=False) print("\n✅ ALL EXPERIMENTS COMPLETED") print("📌 Saved Summary:", summary_path) summary_df
